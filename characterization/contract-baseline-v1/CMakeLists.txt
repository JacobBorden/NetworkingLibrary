cmake_minimum_required(VERSION 3.20)
project(NetworkingContractBaseline LANGUAGES CXX)
include(CTest)
add_subdirectory(../.. legacy)
add_executable(legacy_baseline baseline.cpp)
target_compile_features(legacy_baseline PRIVATE cxx_std_20)
target_include_directories(legacy_baseline PRIVATE ../../src)
target_link_libraries(legacy_baseline PRIVATE SERVER)
target_link_options(legacy_baseline PRIVATE -Wl,--wrap=recv -Wl,--wrap=send)
add_test(NAME legacy_characterization COMMAND legacy_baseline)
set_tests_properties(legacy_characterization PROPERTIES TIMEOUT 10)
add_test(NAME proposed_binary_contract COMMAND legacy_baseline --contract)
set_tests_properties(proposed_binary_contract PROPERTIES TIMEOUT 10 LABELS contract-gap)
